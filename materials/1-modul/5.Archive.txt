#Compression
	gzip filename
	gzip -d filename.gz
	Options (gzip): -d, -k,-r,-v,-N (N=1-9), -h

	extentions : .gz .bz2 .xz .zst 
	options with tar: c f t x v p --exclude
	compression option : z j J --zst
	syntax: 	
		tar --option test.tar test/
		tar -xf test.tar -C /tmp/untarred

	syntax for zip:
		 zip -r test.zip test/
		 unzip test.zip
		 unzip test.zip -d /tmp/unziped

	syntax for rar:
		rar a test.tar test/
		rar x test.tar /tmp/unrarred
   rar e (without path)
