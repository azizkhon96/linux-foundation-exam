	Processes:
		foreground and background
		commands: ps, -f, ax, kill, kill -9, killall, pidof,  ctr+c, ctr+z, &, bg, fg, top, free, htop
		
	Services:
		listing : systemctl list-unit-files, systemctl | grep running, service --status-all
		managing: using systemd and unit.d, start, stop, status, disable , enable, restart, reload
		path: /usr/lib/systemd/system, /etc/systemd/system
		example: 
			[Unit]
			Description=Test service
			Requires=Network.target
			After=Network.target

			[Service]
			Type=simple
			ExecStart=/usr/bin/my-script.sh
		
			[Install]
			WantedBy=multi-user.target



	Copying partitoon table , backup 
		MBR:
			backup and restore:
			sfdisk -d /dev/sda > partition.backup
			sfdisk /dev/sdb <  partition.backup

			copying partition to another disk:
			sfdisk -d /dev/sda | sfdisk /dev/sdb

		GPT:
			Backup and restore:
			sgdisk --backup=table /dev/sda
			sgdisk --load-backup=table /dev/sdb
			sgdisk -G /dev/sdb						#randomize GUID 

			copying partition to another disk:
			sgdisk -R /dev/sda /dev/sdb
			sgdisk -G /dev/sdb
