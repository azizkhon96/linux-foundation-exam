	####	Package Maganers	####

#apt 

	apt 		
        update	 			#menuni yangilash
        upgrade				#
        list --upgradable
        full-upgrade
        search <^pack-name$>
        install <pack-name> 
        remove <pack-name>
        purge <pack-name>
        autoremove
        --purge autoremove
        show <pack-name>
        list 
        list | grep <something>
        list --installed
        depends <pack-name>
        edit-source
		install /path/<package.deb>		#local installation
	
	apt-mark    
            hold <pack-name>
            unhold <pack-name>

#yum or dnf

	yum 	
		install
		remove
		update     = apt upgrade
		list
		search
		info       = apt show
		list installed
		install /path/<package.rpm>		#local installation

	yum install dnf		#install dnf	



#dpkg:
			dpkg -i (--install) appname.deb, -r(--remove),-P(--purge), --help

#rpm
				rpm -i appname.rpm		#install
				rpm -U appname.rpm			#upgrade
				rpm -e appname			#erase
				-q 		#query package exist or not
				-v 		#verbose  	-vh  #show hash marks while installation
				-qi		#query info
